# To copy *.no_annot.csv
while read line; do cp ../../results/acc_batch_*/$line.no_annot.csv no_annot/; done < zorya_acc.txt 
while read line; do cp ../../results/$line.no_annot.csv no_annot/; done < zorya_acc.txt

# To get status.txt
while read acc
do
	if [[ ! -f no_annot/${acc}.no_annot.csv ]]
	then 
		echo $acc Pending >> status.txt
	else
		echo $acc Done >> status.txt
	fi
done < zorya_acc.txt 


# Generate combined no_annot file
head -n 1 "$(ls no_annot/*.csv | head -n 1)" > ${today}_zorya.no_annot.combined.csv
tail -n +2 -q no_annot/*.csv >> ${today}_zorya.no_annot.combined.csv
